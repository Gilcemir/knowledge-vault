Python Data Structures

Cheat Sheet  ·  Common Operations  ·  Time Complexity

List

ORDERED · MUTABLE
Init
l = []
l = [1, 2, 3]
l = list(range(5))     # [0,1,2,3,4]
l = [0] * 5            # [0,0,0,0,0]
Add / Remove
l.append(4)          # add to end      O(1)
l.insert(1, 99)      # insert at idx   O(n)
l.extend([5,6])      # add many        O(k)
l.pop()              # remove last     O(1)
l.pop(0)             # remove at idx   O(n)
l.remove(99)         # remove by val   O(n)
l.clear()            # empty list
Access / Search
l[0]                 # first           O(1)
l[-1]                # last            O(1)
l[1:3]               # slice           O(k)
l.index(99)          # find index      O(n)
99 in l              # membership      O(n)
len(l)               # length          O(1)
Sort / Other
l.sort()             # in-place        O(n log n)
l.sort(reverse=True)
sorted(l)            # new list        O(n log n)
l.reverse()          # in-place        O(n)
l.count(2)           # occurrences     O(n)

Dict (Map)

KEY-VALUE · ORDERED*
Init
d = {}
d = {"a": 1, "b": 2}
d = dict(a=1, b=2)
from collections import defaultdict
d = defaultdict(int)   # default 0
d = defaultdict(list)  # default []
Set / Get / Delete
d["key"] = 10          # set             O(1)
d["key"]               # get (KeyError)  O(1)
d.get("key", 0)        # get w/ default  O(1)
del d["key"]           # remove          O(1)
d.pop("key", None)    # pop w/ default  O(1)
d.clear()             # empty dict
Iterate
for k in d:           # keys
for v in d.values():   # values
for k, v in d.items(): # pairs
"key" in d            # membership      O(1)
len(d)                # size            O(1)
Merge / Other
d.update({"c": 3})    # merge in-place
merged = {**d1, **d2}  # merge new dict
d.setdefault("k", [])  # set if missing

Set

UNIQUE · UNORDERED
Init
s = set()             # empty (not {}!)
s = {1, 2, 3}
s = set([1, 2, 2, 3])  # {1,2,3}
Add / Remove
s.add(4)              # add one         O(1)
s.update([5,6])       # add many        O(k)
s.remove(4)           # rm (KeyError)   O(1)
s.discard(4)          # rm (safe)       O(1)
s.pop()               # remove random   O(1)
s.clear()             # empty set
Set Operations
a | b               # union
a & b               # intersection
a - b               # difference
a ^ b               # symmetric diff
a.issubset(b)       # a ⊆ b
a.issuperset(b)    # a ⊇ b
x in s              # membership      O(1)

Queue (deque)

FIFO
Usage
from collections import deque
q = deque()

q.append(1)         # enqueue right O(1)
q.appendleft(0)    # enqueue left  O(1)
q.popleft()         # dequeue left  O(1)
q.pop()             # dequeue right O(1)
q[0]                # peek front    O(1)
q[-1]               # peek back     O(1)
len(q) == 0         # is empty

Stack (deque)

THREAD-SAFE
Prefer deque over list for performance-critical stacks
from collections import deque
stack = deque()

stack.append(1)      # push      O(1)
stack.append(2)
top = stack[-1]     # peek      O(1)
val = stack.pop()   # pop       O(1)

# deque is O(1) on both ends
# list.pop(0) is O(n) !

Heap / Priority Queue

MIN-HEAP BY DEFAULT
Min-Heap
import heapq

h = []
heapq.heappush(h, 3)  # O(log n)
heapq.heappush(h, 1)
heapq.heappush(h, 2)

top = h[0]             # peek min  O(1)
val = heapq.heappop(h) # pop min   O(log n)

# heapify existing list
nums = [3, 1, 4, 1, 5]
heapq.heapify(nums)   # O(n)
Max-Heap (negate values)
h = []
heapq.heappush(h, -3) # negate!
heapq.heappush(h, -1)
top = -h[0]            # peek max  O(1)
val = -heapq.heappop(h)# pop max

# Tuples: (priority, value)
heapq.heappush(h, (1, "low"))
heapq.heappush(h, (5, "high"))
pri, val = heapq.heappop(h)

# n largest / smallest
heapq.nlargest(3, nums)
heapq.nsmallest(3, nums)

Tuple

IMMUTABLE
Init & Access
t = (1, 2, 3)
t = 1, 2, 3          # same thing
t = (42,)             # single element!

t[0]                  # access      O(1)
a, b, c = t           # unpack
a, *rest = t          # star unpack
x, y = y, x           # swap vars

Counter

FREQUENCY MAP
Usage
from collections import Counter

c = Counter([1,2,2,3])
# Counter({2:2, 1:1, 3:1})
c = Counter("abracadabra")

c["a"]               # count of "a"
c.most_common(3)     # top 3 elements
c.total()            # sum of counts
c1 + c2              # combine counts
