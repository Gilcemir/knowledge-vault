String Basics
Create & concat
s = "Hello, World!"
s2 = 'single quotes'
multi = """multi
line"""

Slicing & indexing
s = "Python"
s[0]       # 'P'
s[-1]      # 'n'
s[1:4]    # 'yth'
s[::-1]   # 'nohtyP' (reversed)

String Methods
Case & whitespace
s = "  Hello World  "
s.upper()        # "  HELLO WORLD  "
s.lower()        # "  hello world  "
s.title()        # "  Hello World  "
s.strip()        # "Hello World"
s.lstrip()       # "Hello World  "
s.rstrip()       # "  Hello World"
Search & replace
s = "banana"
s.find("an")      # 1  (first occurrence)
s.count("a")     # 3
s.replace("a", "o")   # "bonono"
s.startswith("ban") # True
s.endswith("na")   # True
"an" in s         # True
Padding & align
"42".zfill(5)       # "00042"
"hi".center(10, "-")  # "----hi----"
"hi".ljust(10, ".")   # "hi........"
"hi".rjust(10, ".")   # "........hi"
✂️
Split & Join
split()
s = "one,two,three"
s.split(",")       # ['one','two','three']
s.split(",", 1)    # ['one','two,three']

"  a  b  c  ".split() # ['a','b','c']

# splitlines — split on newlines
"a\nb\nc".splitlines()  # ['a','b','c']
join()
words = ["Hello", "World"]
" ".join(words)     # "Hello World"
"-".join(words)     # "Hello-World"
"".join(words)      # "HelloWorld"

# join + split round-trip
csv = ",".join(["a","b","c"])  # "a,b,c"
csv.split(",")            # ['a','b','c']
partition()
"user@example.com".partition("@")
# ('user', '@', 'example.com')
"key=value".partition("=")
# ('key', '=', 'value')

Type Checks
isinstance & type
isinstance(42, int)          # True
isinstance(3.14, float)      # True
isinstance("hi", str)        # True
isinstance([], list)         # True
isinstance({}, dict)         # True
isinstance(42, (int, float))  # True (either)
type(42) is int              # True
String character checks
"abc123".isalnum()   # True  — letters+digits
"abc".isalpha()      # True  — letters only
"123".isdigit()      # True  — digits only
"123".isnumeric()    # True  — numeric chars
"  ".isspace()       # True  — whitespace
"Hello".istitle()    # True  — title case
"ABC".isupper()      # True
"abc".islower()      # True
Safe number parsing
def is_number(s):
    try:
        float(s)
        return True
    except ValueError:
        return False

is_number("3.14")   # True
is_number("abc")    # False
📋
Lists
Create & mutate
lst = [3, 1, 4, 1, 5]
lst.append(9)         # [3,1,4,1,5,9]
lst.insert(0, 99)      # [99,3,1,4,1,5,9]
lst.extend([7, 8])     # adds [7,8] to end
lst.remove(1)         # removes first 1
lst.pop()             # removes & returns last
lst.pop(0)            # removes & returns first
Search & sort
lst = [3, 1, 4, 2]
lst.index(4)    # 2  (position of 4)
lst.count(1)    # 1  (how many 1s)
4 in lst       # True
lst.sort()     # [1,2,3,4] in-place
lst.reverse()  # [4,3,2,1] in-place
sorted(lst)    # returns new sorted list
List comprehensions
squares = [x**2 for x in range(5)]
# [0, 1, 4, 9, 16]

evens = [x for x in range(10) if x % 2 == 0]
# [0, 2, 4, 6, 8]

flat = [x for row in [[1,2],[3,4]] for x in row]
# [1, 2, 3, 4]

Dictionaries
Create & access
d = {"name": "Alice", "age": 30}
d["name"]              # "Alice"
d.get("age")           # 30
d.get("city", "N/A")   # "N/A" (default)
d["email"] = "a@x.com"  # add/update key
del d["age"]           # delete key
Iterate & views
d.keys()    # dict_keys(['name','email'])
d.values()  # dict_values(['Alice','a@x.com'])
d.items()   # dict_items([('name','Alice')…])

for k, v in d.items():
    print(f"{k}: {v}")
Dict comprehension & merge
sq = {x: x**2 for x in range(4)}
# {0:0, 1:1, 2:4, 3:9}

# merge (Python 3.9+)
merged = {"a": 1} | {"b": 2}
# {'a': 1, 'b': 2}

# merge (older style)
merged = {**d1, **d2}

Regular Expressions
Search & match
import re

re.search(r"\d+", "abc123")   # match
re.match(r"\d+", "123abc")    # start only
re.fullmatch(r"\d+", "123")  # whole str
re.findall(r"\d+", "a1b22")  # ['1','22']
Sub & split
re.sub(r"\s+", "-", "a b  c")
# "a-b-c"

re.split(r"[,;]", "a,b;c")
# ['a', 'b', 'c']

# Compile for reuse
pat = re.compile(r"[A-Z]\w+")
pat.findall("Hello World")
# ['Hello', 'World']
Common patterns
# email validation
pat = r"[\w.-]+@[\w.-]+\.\w{2,}"

# digits only
pat = r"^\d+$"

# URL
pat = r"https?://[^\s]+"

# strip HTML tags
re.sub(r"<[^>]+>", "", "<b>bold</b>")

Sorting & Ordering
sorted() & sort()
nums = [3, 1, 4, 1, 5]
sorted(nums)               # [1,1,3,4,5]
sorted(nums, reverse=True) # [5,4,3,1,1]

words = ["banana", "apple", "fig"]
sorted(words)               # alpha
sorted(words, key=len)     # by length
key= functions
people = [
    {"name": "Bob",   "age": 30},
    {"name": "Alice", "age": 25},
]
sorted(people, key=lambda x: x["age"])

from operator import itemgetter
sorted(people, key=itemgetter("name"))
min / max / sum
data = [3, 1, 4, 1, 5]
min(data)   # 1
max(data)   # 5
sum(data)   # 14
sum(data) / len(data)  # 2.8 (avg)

Useful Builtins
enumerate & zip
fruits = ["apple", "banana"]
for i, v in enumerate(fruits):
    print(i, v)   # 0 apple  1 banana

names  = ["Alice", "Bob"]
scores = [95, 82]
for n, s in zip(names, scores):
    print(f"{n}: {s}")
map & filter
nums = [1, 2, 3, 4]
list(map(str, nums))
# ['1','2','3','4']

list(filter(lambda x: x > 2, nums))
# [3, 4]

# any / all
any(x > 3 for x in nums)  # True
all(x > 0 for x in nums)  # True
Counter & defaultdict
from collections import Counter, defaultdict

Counter("banana")
# Counter({'a':3,'n':2,'b':1})

dd = defaultdict(list)
dd["x"].append(1)    # no KeyError

Quick Reference — All String Methods
.upper() / .lower()Change case
.title() / .capitalize()Title / first-letter case
.strip() / .lstrip() / .rstrip()Remove whitespace
.split(sep, maxsplit)Split into list
.splitlines()Split on newlines
sep.join(iterable)Join list → string
.replace(old, new, count)Replace substrings
.find(sub) / .rfind(sub)Index of substring (−1 if not found)
.index(sub)Like find but raises ValueError
.count(sub)Count non-overlapping occurrences
.startswith(prefix)Boolean prefix check
.endswith(suffix)Boolean suffix check
.isalpha()All alphabetic characters
.isdigit()All digit characters
.isnumeric()All numeric (incl. fractions, superscripts)
.isalnum()All alphanumeric
.isspace()All whitespace
.isupper() / .islower()All upper / lower case
.istitle()Title-case string
.zfill(width)Zero-pad numeric string
.center(w, fill)Center in field
.ljust(w) / .rjust(w)Left / right justify
.encode(enc)Encode to bytes
.format(*args, **kw)Old-style string formatting
